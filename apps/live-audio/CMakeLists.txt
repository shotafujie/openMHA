cmake_minimum_required(VERSION 3.21)
project(OpenMHALive VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENMHA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.local" CACHE PATH "openMHA installation")
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
find_path(MHA_INCLUDE mha_toolbox.h PATHS "${OPENMHA_ROOT}/include/openmha" "${OPENMHA_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(MHA_LIBRARY NAMES openmha PATHS "${OPENMHA_ROOT}/lib" "${OPENMHA_ROOT}/bin" REQUIRED NO_DEFAULT_PATH)
find_path(PA_INCLUDE portaudio.h REQUIRED)
find_library(PA_LIBRARY NAMES portaudio portaudio_static REQUIRED)
qt_add_executable(OpenMHALive MACOSX_BUNDLE main.cpp engine.cpp)
target_include_directories(OpenMHALive SYSTEM PRIVATE "${MHA_INCLUDE}" "${PA_INCLUDE}")
target_link_libraries(OpenMHALive PRIVATE Qt6::Widgets "${MHA_LIBRARY}" "${PA_LIBRARY}")
if(WIN32)
    set(MHA_PLUGIN_DIR "${OPENMHA_ROOT}/bin")
else()
    set(MHA_PLUGIN_DIR "${OPENMHA_ROOT}/lib")
endif()
target_compile_definitions(OpenMHALive PRIVATE MHA_DEFAULT_PATH="${MHA_PLUGIN_DIR}")
set_target_properties(OpenMHALive PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.openmha.fork.live" BUILD_RPATH "${OPENMHA_ROOT}/lib")
enable_testing()
add_test(NAME live_dsp COMMAND OpenMHALive --self-test)
set_tests_properties(live_dsp PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
